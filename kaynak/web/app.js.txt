function h(tag, props = {}, ...kids) {
  const e = document.createElement(tag);
  for (const [k, v] of Object.entries(props)) {
    if (k === 'class') e.className = v; else if (k === 'text') e.textContent = v; else if (k.startsWith('on')) e.addEventListener(k.slice(2), v);
    else if (v === true) e.setAttribute(k, ''); else if (v !== false && v != null) e.setAttribute(k, v);
  }
  kids.flat().forEach((c) => { if (c != null) e.append(c.nodeType ? c : document.createTextNode(String(c))); });
  return e;
}

function sayiAnim(el, hedef, yaz) {
  if (!el) return;
  const bas = typeof el._v === 'number' ? el._v : 0; el._v = hedef;
  if (el._raf) cancelAnimationFrame(el._raf); clearTimeout(el._son);
  if (hareketAz() || bas === hedef || typeof requestAnimationFrame !== 'function') { yaz(hedef); return; }
  const t0 = performance.now();
  const adim = () => { const k = Math.max(0, Math.min(1, (performance.now() - t0) / SAYIM_MS)), e = 1 - Math.pow(1 - k, 3); yaz(Math.round(bas + (hedef - bas) * e)); if (k < 1) el._raf = requestAnimationFrame(adim); };
  el._raf = requestAnimationFrame(adim);
  el._son = setTimeout(() => { cancelAnimationFrame(el._raf); yaz(hedef); }, SAYIM_MS + 30);   // kare atlanırsa bile 0,43 sn'de doğru değer
}

// ---------- Sohbet arka planı: yalnızca yazı karakterleriyle çizilen kara delik ----------
// Ortada boş karanlık disk, çevresinde parlak harf halkası, eğik ve yavaş dönen bir birikim diski. Süs olduğu için aria-hidden bir tuvale çizilir.
// Dönen değer çizimi durdurur. Hareket azaltma tercihinde tek durağan kare çizilir; sekme gizliyken çizim yapılmaz.
function karaDelik(tuval) {
  const c = tuval.getContext && tuval.getContext('2d'); if (!c) return () => {};
  const HALKA = 'OZMIRA', TOZ = '.,:;~+=*xo08#', EN = 7, BOY = 12, ALTIN = '#f3b84b', BEYAZ = '#fff2d8', EGIM = 0.24, CA = Math.cos(-0.16), SA = Math.sin(-0.16);
  const sabit = typeof matchMedia === 'function' && matchMedia('(prefers-reduced-motion: reduce)').matches;
  let w = 0, yuk = 0, kare = 0, son = -1e9, bitti = false;
  function olc() {
    const r = tuval.getBoundingClientRect(), o = Math.min(2, window.devicePixelRatio || 1); w = Math.round(r.width); yuk = Math.round(r.height);
    tuval.width = Math.max(1, Math.round(w * o)); tuval.height = Math.max(1, Math.round(yuk * o));
    c.setTransform(o, 0, 0, o, 0, 0); c.font = '11px ui-monospace, "Cascadia Mono", Menlo, Consolas, monospace'; c.textAlign = 'center'; c.textBaseline = 'middle';
  }
  function ciz(z) {
    c.clearRect(0, 0, w, yuk); if (w < 80 || yuk < 80) return;
    const cx = w / 2, cy = yuk / 2, r0 = Math.min(yuk * 0.27, w * 0.135), ic = r0 * 1.55, dis = r0 * 3.6, halka = r0 + Math.max(BOY * 1.5, r0 * 0.24);
    const i0 = Math.max(0, Math.floor((cx - dis) / EN)), i1 = Math.min(Math.ceil(w / EN), Math.ceil((cx + dis) / EN));
    for (let j = 0; j * BOY < yuk; j++) for (let i = i0; i < i1; i++) {
      const px = i * EN + EN / 2, py = j * BOY + BOY / 2, dx = px - cx, dy = py - cy, r = Math.hypot(dx, dy);
      const u = dx * CA - dy * SA, v = (dx * SA + dy * CA) / EGIM, ro = Math.hypot(u, v);   // eğik diskin kendi düzlemindeki konum
      let p = 0, kr = '', renk = ALTIN;
      if (ro > ic && ro < dis && (r > halka || v > 0)) {   // disk: arka yarısı deliğin ardında kalır, ön yarısı önünden geçer
        const fi = Math.atan2(v, u), hiz = 0.45 * Math.pow(ic / ro, 1.5);
        p = Math.pow(1 - (ro - ic) / (dis - ic), 1.1) * (0.5 + 0.5 * (0.5 + 0.5 * Math.sin(3 * (fi - hiz * z) + ro * 0.07))) * (1 - 0.18 * Math.cos(fi));
        if (p < 0.14 && (i * 7 + j * 13) % 3) p = 0;   // dış kenar seyrekleşerek söner
        kr = TOZ[Math.min(TOZ.length - 1, Math.floor(p * TOZ.length))]; if (p > 0.62) renk = BEYAZ;
      }
      if (p < 0.14 && r >= r0 && r <= halka) {             // ışık halkası: marka harfleri, yavaşça döner
        const n = HALKA.length * 5, sira = Math.floor(((Math.atan2(dy, dx) / (2 * Math.PI) + 1 + z * 0.012) % 1) * n);
        kr = HALKA[sira % HALKA.length]; p = 0.8 + 0.2 * Math.sin(z * 0.9 + sira); renk = BEYAZ;
      }
      if (p <= 0.04 || !kr) continue;
      c.globalAlpha = Math.min(1, p); c.fillStyle = renk; c.fillText(kr, px, py);
    }
    c.globalAlpha = 1;
  }
  function durdur() { bitti = true; cancelAnimationFrame(kare); if (gozcu) gozcu.disconnect(); }
  function adim(ts) {
    if (bitti) return; if (!tuval.isConnected) { durdur(); return; }
    kare = requestAnimationFrame(adim);
    if (document.hidden || ts - son < 80) return;        // en çok saniyede 12 kare
    son = ts; ciz(ts / 1000);
  }
  const gozcu = typeof ResizeObserver === 'function' ? new ResizeObserver(() => { if (!bitti) { olc(); ciz(sabit ? 6 : performance.now() / 1000); } }) : null;
  if (gozcu) gozcu.observe(tuval); else { olc(); ciz(6); }
  if (!sabit) kare = requestAnimationFrame(adim);
  return durdur;
}

    const denenebilir = d.eylem && !(d.gercek && ornekMi);
    pencereAc([
      h('h1', { text: t('r.baslik') }), h('p', { class: 'lead', text: t('r.lead') + ' ' + t('r.ilerleme', { n: biten.size, m: DERSLER.length }) }),
      h('div', { class: 'guide' },
        h('nav', { 'aria-label': t('r.dersler') }, h('h3', { text: t('r.temel') }), grup(false), h('h3', { text: t('r.ileri') }), grup(true)),
        h('article', {}, h('h2', { tabindex: '-1', text: (sira + 1) + '. ' + t('r.' + d.k + '.b') }), paragraflar,
          h('div', { class: 'row' },
            denenebilir ? h('button', { class: 'btn main', type: 'button', text: t('r.dene'), onclick: () => { biten.add(d.k); rehberYaz(biten); d.eylem(); } }) : null,
            h('button', { class: 'btn', type: 'button', text: biten.has(d.k) ? t('r.tamam') : t('r.isaretle'), disabled: biten.has(d.k), onclick: () => { biten.add(d.k); rehberYaz(biten); ciz('sonraki'); } }),
            sira < DERSLER.length - 1 ? h('button', { class: 'btn', type: 'button', 'data-o': 'sonraki', text: t('r.sonraki'), onclick: () => { biten.add(d.k); rehberYaz(biten); secili = DERSLER[sira + 1].k; ciz('ders'); } }) : null))),
      h('div', { class: 'foot' }, h('span'), h('button', { class: 'btn main', type: 'button', text: t('c.kapat'), onclick: pencereKapat })),
    ], true);
    const p = $('panel'), hedef = odak === 'sonraki' ? (anahtarli(p, 'sonraki') || p.querySelector('article h2')) : odak === 'ders' ? p.querySelector('article h2') : null;
    if (hedef) hedef.focus();
  }
  ciz();
}

async function yukle(zorla) {
  let d;
  try { d = await api('GET', '/api/durum'); } catch (_) { if (durum) sideCiz(true); return; }
  if (d.yok) {
    if (!ornekMi || !durum || zorla) { ornekMi = true; durum = ORNEK; demoAcik = true; sahne.setDurum(durum); sahne.demo(true); dockSig = ''; hepsiniCiz(); }
    return;
  }
  const ilkGercek = ornekMi || zorla; ornekMi = false; durum = d; sahne.setDurum(d);
  if (ilkGercek) { demoAcik = false; sahne.demo(false); dockSig = ''; }
  hepsiniCiz();
}

async function boot() {
  uygula();
  await hesapYukle();
  sahne = createSahne($('stage'));
  sahne.onSelect(kartCiz);
  durum = ORNEK; sahne.setDurum(ORNEK); sahne.demo(true); hepsiniCiz();
  await yukle(false);
  if (ornekMi) sihirbaz();
  poll = setInterval(() => { if (!document.hidden && $('modal').hidden) yukle(false); }, 2000);
}
boot().catch(fail);
